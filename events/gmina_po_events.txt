#prepare_sports_events_aspect
country_event = {
	id = po_events.6
	title = po_events.6.t
	desc = po_events.6.d
	picture = PO_ASPECT_Picture

	is_triggered_only = yes

	option = {
		name = po_events.6.ab

		highlight = yes

		trigger = {

			OR = {
				any_owned_province = {
					OR = {
						NOT = { development = 30 }
						NOT = { has_terrain = urban } #tooltip reasons
						NOT = { has_terrain = high_urban } 
					}
	
				}
	
				NOT = {
					had_country_flag = {
						flag = sport_reform_long_term_solution
						days = 1825
					}
				}
			}
			
		}

		add_church_power = 100

		ai_chance = { factor = 0 }

	}

	option = {
		name = po_events.6.a

		trigger = {
			any_owned_province = {
				OR = {
					development = 30
					has_terrain = urban
					has_terrain = high_urban
				}
			}

			had_country_flag = {
				flag = sport_reform_long_term_solution
				days = 1825
			}
			
		}

		random_owned_province = {
			limit = {
				OR = {
					development = 30
					has_terrain = urban
					has_terrain = high_urban
				}
			}
			add_province_modifier = { 
				name = stadium_infrastructure
				duration = -1  
			}
		}


		hidden_effect = {
			set_country_flag = sport_reform_long_term_solution
		}


		ai_chance = { factor = 10 }

	}

	option = {
		name = po_events.6.b

		trigger = {
			NOT = {
				has_country_modifier = orlik_2012
			}
		}

		add_country_modifier = { 
			name = orlik_2012
			duration = 1825
		}

		hidden_effect = {
			clr_country_flag = sport_reform_medium_term_solution
			set_country_flag = sport_reform_medium_term_solution
		}
		
		
		ai_chance = { factor = 40 }

	}

	option = {
		name = po_events.6.c

		add_estate_loyalty_modifier = {
			estate = estate_families
			desc = TBD
			loyalty = 10
			duration = 1825
		}

		add_estate_loyalty_modifier = {
			estate = estate_retirees
			desc = TBD
			loyalty = 10
			duration = 1825
		}

		add_estate_loyalty_modifier = {
			estate = estate_middle_class

		}
