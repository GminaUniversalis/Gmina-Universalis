defined_text = {
	name = Getcommunism_pzprAvailability

	text = {
		localisation_key = communism_pzprAvailable

		trigger = {
			is_religion_enabled = communism_pzpr
		}
	}

	text = {
		localisation_key = communism_pzprAvailableNot

		trigger = {
			NOT = { is_religion_enabled = communism_pzpr }
		}
	}

	text = {
		localisation_key = communism_pzprAvailableNotExists

		trigger = {
			is_religion_enabled = communism_pzpr
			1500 = {
				check_variable = {
					which = communism_pzpr_city
					value = 0
				}
				NOT = {
					check_variable = {
						which = communism_pzpr_city
						value = 1
					}
				}
			}

		}
	}

}
